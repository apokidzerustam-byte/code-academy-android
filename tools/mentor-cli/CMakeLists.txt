# Сборка CLI наставника на компьютере/CI (Linux): то же ядро, что в приложении.
cmake_minimum_required(VERSION 3.22.1)
project(mentor_cli LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SRC "${CMAKE_CURRENT_LIST_DIR}/../../app/src/main/cpp")
set(LLAMA_DIR "${SRC}/llama.cpp" CACHE PATH "исходники llama.cpp")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(opt LLAMA_CURL LLAMA_OPENSSL LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP)
    set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_DIR}" build-llama)
add_executable(mentor_cli "${SRC}/mentor_cli.cpp" "${SRC}/mentor_core.cpp")
target_include_directories(mentor_cli PRIVATE "${LLAMA_DIR}/include" "${LLAMA_DIR}/ggml/include")
target_link_libraries(mentor_cli llama ggml)
